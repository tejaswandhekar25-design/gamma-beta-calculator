      Gamma Function in MATLAB

      % Gamma function using built-in
n = input('Enter n: ');
result = gamma(n);
fprintf('Γ(%g) = %g\n', n, result);

% Gamma via factorial for positive integers
if n == floor(n) && n > 0
    result = factorial(n - 1);
    fprintf('Γ(%d) = (%d)! = %d\n', n, n-1, result);
end

% Symbolic computation
syms t
f = t^(n-1) * exp(-t);
result = int(f, t, 0, inf);
disp(result);

      Beta Function in MATLAB

      % Beta function using built-in
x = input('Enter x: ');
y = input('Enter y: ');
result = beta(x, y);
fprintf('B(%g, %g) = %g\n', x, y, result);

% Beta via Gamma functions
result = gamma(x)*gamma(y)/gamma(x+y);
fprintf('B(%g,%g) = Γ(%g)Γ(%g)/Γ(%g) = %g\n', ...
        x, y, x, y, x+y, result);

% Symbolic integral
syms t
f = t^(x-1) * (1-t)^(y-1);
result = int(f, t, 0, 1);
disp(result);
